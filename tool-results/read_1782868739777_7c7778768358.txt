     1→'use client';
     2→
     3→import { useRef, useEffect, useMemo, useCallback, useState } from 'react';
     4→import { Note, TAG_COLORS, EmbedBlock, MindMapBlock, KanbanEmbedBlock, TodoEmbedBlock } from '@/types';
     5→import { renderMarkdownOverlay, formatDate, countWords } from '@/utils/markdown';
     6→import { renderMarkdownHtml } from '@/utils/markdownHtml';
     7→import { computeDiff } from '@/utils/diff';
     8→import { useEditor, ViewMode } from '@/hooks/useEditor';
     9→import { ArrowLeft, RefreshCw, Eye, Pencil, GitCompare, Network, CheckSquare, KanbanSquare } from 'lucide-react';
    10→import { FormattingToolbar } from './FormattingToolbar';
    11→import { MindMapEmbed } from './MindMapEmbed';
    12→import { TodoEmbed } from './TodoEmbed';
    13→import { KanbanEmbed } from './KanbanEmbed';
    14→import { SlashMenu, SlashCommand } from './SlashMenu';
    15→import {
    16→  parseBodySegments, replaceEmbedInBody, removeEmbedFromBody,
    17→  createMindMapBlock, createTodoEmbedBlock, createKanbanEmbedBlock,
    18→  serializeEmbed,
    19→} from '@/utils/embeds';
    20→
    21→interface EditorCanvasProps {
    22→  note: Note;
    23→  allNotes: Note[];
    24→  dirty: boolean;
    25→  editor: ReturnType<typeof useEditor>;
    26→  onSave: (id: string, patch: Partial<Note>) => void;
    27→  onOpenNote: (id: string) => void;
    28→  onOpenNoteByTitle: (title: string) => void;
    29→  onToggleEvergreen: (id: string) => void;
    30→  onOpenKanbanPage?: () => void;
    31→}
    32→
    33→export function EditorCanvas({
    34→  note,
    35→  allNotes,
    36→  dirty,
    37→  editor,
    38→  onSave,
    39→  onOpenNote,
    40→  onOpenNoteByTitle,
    41→  onToggleEvergreen,
    42→  onOpenKanbanPage,
    43→}: EditorCanvasProps) {
    44→  const textareaRef = useRef<HTMLTextAreaElement>(null);
    45→  const preRef = useRef<HTMLPreElement>(null);
    46→  const [slashOpen, setSlashOpen] = useState(false);
    47→  const [slashPos, setSlashPos] = useState<{ x: number; y: number } | null>(null);
    48→
    49→  // Auto-resize textarea to match content height
    50→  const autoResize = useCallback(() => {
    51→    const ta = textareaRef.current;
    52→    if (!ta) return;
    53→    ta.style.height = 'auto';
    54→    ta.style.height = ta.scrollHeight + 'px';
    55→  }, []);
    56→
    57→  useEffect(() => {
    58→    autoResize();
    59→  }, [editor.body, note?.id, autoResize]);
    60→
    61→  // Re-measure on window resize
    62→  useEffect(() => {
    63→    const handler = () => autoResize();
    64→    window.addEventListener('resize', handler);
    65→    return () => window.removeEventListener('resize', handler);
    66→  }, [autoResize]);
    67→
    68→  // Render markdown to HTML overlay (for edit mode)
    69→  const overlayHtml = useMemo(() => renderMarkdownOverlay(editor.body), [editor.body]);
    70→
    71→  // Render markdown to full HTML (for preview mode)
    72→  const previewHtml = useMemo(() => renderMarkdownHtml(editor.body), [editor.body]);
    73→
    74→  // Parse body into text + embed segments
    75→  const segments = useMemo(() => parseBodySegments(editor.body), [editor.body]);
    76→
    77→  // Update an embed block in the body
    78→  const updateEmbed = useCallback((blockId: string, newBlock: EmbedBlock) => {
    79→    const nextBody = replaceEmbedInBody(editor.body, blockId, newBlock);
    80→    editor.updateBody(nextBody);
    81→  }, [editor]);
    82→
    83→  // Remove an embed block from the body
    84→  const removeEmbed = useCallback((blockId: string) => {
    85→    const nextBody = removeEmbedFromBody(editor.body, blockId);
    86→    editor.updateBody(nextBody);
    87→  }, [editor]);
    88→
    89→  // Insert an embed at the cursor position (or end of body)
    90→  const insertEmbedAtCursor = useCallback((block: EmbedBlock) => {
    91→    const serialized = serializeEmbed(block);
    92→    const ta = textareaRef.current;
    93→    const body = editor.body;
    94→    if (ta) {
    95→      const pos = ta.selectionStart;
    96→      const before = body.slice(0, pos);
    97→      const after = body.slice(pos);
    98→      const prefix = before.length > 0 && !before.endsWith('\n') ? '\n\n' : '';
    99→      const suffix = after.length > 0 && !after.startsWith('\n') ? '\n\n' : '';
   100→      editor.updateBody(before + prefix + serialized + suffix + after);
   101→    } else {
   102→      editor.updateBody(body + '\n\n' + serialized);
   103→    }
   104→  }, [editor]);
   105→
   106→  // Slash command handlers — defined as plain array (recomputed each render, cheap)
   107→  const slashCommands: SlashCommand[] = [
   108→    {
   109→      id: 'mindmap',
   110→      label: 'mindmap',
   111→      description: 'embed an interactive mind map',
   112→      icon: Network,
   113→      action: () => { insertEmbedAtCursor(createMindMapBlock(note.title)); setSlashOpen(false); },
   114→    },
   115→    {
   116→      id: 'todo',
   117→      label: 'todo',
   118→      description: 'embed a lightweight checklist',
   119→      icon: CheckSquare,
   120→      action: () => { insertEmbedAtCursor(createTodoEmbedBlock()); setSlashOpen(false); },
   121→    },
   122→    {
   123→      id: 'kanban',
   124→      label: 'kanban',
   125→      description: 'embed a kanban board preview',
   126→      icon: KanbanSquare,
   127→      action: () => { insertEmbedAtCursor(createKanbanEmbedBlock()); setSlashOpen(false); },
   128→    },
   129→  ];
   130→
   131→  // Detect "/" at start of line or after whitespace → open slash menu
   132→  const handleSlashDetect = useCallback((e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   133→    if (e.key !== '/') return;
   134→    const ta = e.currentTarget;
   135→    const pos = ta.selectionStart;
   136→    const before = ta.value.slice(0, pos);
   137→    // Check if '/' is at start of line or after whitespace
   138→    const isAtLineStart = before.length === 0 || before.endsWith('\n') || before.endsWith(' ');
   139→    if (!isAtLineStart) return;
   140→
   141→    // Get caret coordinates
   142→    const rect = ta.getBoundingClientRect();
   143→    // Approximate position — in a real implementation we'd measure caret position
   144→    // For now, position below the textarea top + some offset
   145→    setSlashPos({ x: rect.left + 48, y: rect.top + 40 + (ta.selectionStart / 80) * 18 });
   146→    setSlashOpen(true);
   147→  }, []);
   148→
   149→  const handleSlashSelect = useCallback((cmd: SlashCommand) => {
   150→    cmd.action();
   151→    setSlashOpen(false);
   152→  }, []);
   153→
   154→  // Render an embed block as a React component
   155→  const renderEmbed = (block: EmbedBlock) => {
   156→    if (block.type === 'mindmap') {
   157→      return (
   158→        <MindMapEmbed
   159→          block={block as MindMapBlock}
   160→          onUpdate={(updated) => updateEmbed(block.id, updated)}
   161→          onRemove={() => removeEmbed(block.id)}
   162→          onOpenNoteByTitle={onOpenNoteByTitle}
   163→        />
   164→      );
   165→    }
   166→    if (block.type === 'todo') {
   167→      return (
   168→        <TodoEmbed
   169→          block={block as TodoEmbedBlock}
   170→          onUpdate={(updated) => updateEmbed(block.id, updated)}
   171→          onRemove={() => removeEmbed(block.id)}
   172→          onOpenNote={onOpenNote}
   173→          notes={allNotes}
   174→        />
   175→      );
   176→    }
   177→    if (block.type === 'kanban') {
   178→      return (
   179→        <KanbanEmbed
   180→          block={block as KanbanEmbedBlock}
   181→          onRemove={() => removeEmbed(block.id)}
   182→          onOpenFullBoard={() => onOpenKanbanPage?.()}
   183→        />
   184→      );
   185→    }
   186→    return null;
   187→  };
   188→
   189→  // Wiki-link click handling: clicks pass through the transparent textarea
   190→  // only when the underlying <pre> has a data-wiki span at that location.
   191→  // Approach: handle clicks on the <pre> itself, but since textarea is on top,
   192→  // we use mousedown coordinates to detect wiki-link targets by walking the
   193→  // overlay DOM. To keep this simple, we add a click handler on the wrap and
   194→  // check the underlying element via document.elementFromPoint.
   195→  const handleWrapClick = useCallback(
   196→    (e: React.MouseEvent) => {
   197→      // Only intercept if user wasn't selecting text
   198→      const sel = window.getSelection();
   199→      if (sel && sel.toString().length > 0) return;
   200→      // Only intercept Ctrl/Cmd-click to open wiki-link, so normal clicks
   201→      // place the caret as expected.
   202→      if (!e.metaKey && !e.ctrlKey) return;
   203→
   204→      const target = e.target as HTMLElement;
   205→      // Hide textarea temporarily to get element under cursor
   206→      const ta = textareaRef.current;
   207→      if (!ta) return;
   208→      const prevPointer = ta.style.pointerEvents;
   209→      ta.style.pointerEvents = 'none';
   210→      const el = document.elementFromPoint(e.clientX, e.clientY) as HTMLElement | null;
   211→      ta.style.pointerEvents = prevPointer;
   212→
   213→      const wikiEl = el?.closest('[data-wiki]') as HTMLElement | null;
   214→      if (wikiEl) {
   215→        const title = wikiEl.getAttribute('data-wiki');
   216→        if (title) {
   217→          e.preventDefault();
   218→          onOpenNoteByTitle(title);
   219→        }
   220→      }
   221→    },
   222→    [onOpenNoteByTitle],
   223→  );
   224→
   225→  const wrapSelection = useCallback(
   226→    (prefix: string, suffix: string, placeholder: string = 'text') => {
   227→      const ta = textareaRef.current;
   228→      if (!ta) return;
   229→      const start = ta.selectionStart;
   230→      const end = ta.selectionEnd;
   231→      const selected = editor.body.slice(start, end);
   232→      const text = selected || placeholder;
   233→      const inserted = prefix + text + suffix;
   234→      const next = editor.body.slice(0, start) + inserted + editor.body.slice(end);
   235→      editor.updateBody(next);
   236→      requestAnimationFrame(() => {
   237→        if (selected) {
   238→          ta.selectionStart = start;
   239→          ta.selectionEnd = start + inserted.length;
   240→        } else {
   241→          ta.selectionStart = start + prefix.length;
   242→          ta.selectionEnd = start + prefix.length + placeholder.length;
   243→        }
   244→        ta.focus();
   245→      });
   246→    },
   247→    [editor],
   248→  );
   249→
   250→  const handleKeyDown = useCallback(
   251→    (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   252→      const meta = e.metaKey || e.ctrlKey;
   253→      if (e.key === 'Tab') {
   254→        e.preventDefault();
   255→        const ta = e.currentTarget;
   256→        const start = ta.selectionStart;
   257→        const end = ta.selectionEnd;
   258→        const next = editor.body.slice(0, start) + '  ' + editor.body.slice(end);
   259→        editor.updateBody(next);
   260→        requestAnimationFrame(() => {
   261→          ta.selectionStart = ta.selectionEnd = start + 2;
   262→        });
   263→        return;
   264→      }
   265→      // Formatting shortcuts
   266→      if (meta && e.key.toLowerCase() === 'b') {
   267→        e.preventDefault();
   268→        wrapSelection('**', '**', 'bold text');
   269→      } else if (meta && e.key.toLowerCase() === 'i') {
   270→        e.preventDefault();
   271→        wrapSelection('*', '*', 'italic text');
   272→      } else if (meta && e.key.toLowerCase() === 'u') {
   273→        e.preventDefault();
   274→        wrapSelection('<u>', '</u>', 'underlined');
   275→      } else if (meta && e.shiftKey && e.key.toLowerCase() === 'k') {
   276→        // Cmd+Shift+K — inline code (Cmd+K alone is the command palette)
   277→        e.preventDefault();
   278→        wrapSelection('`', '`', 'code');
   279→      }
   280→    },
   281→    [editor, wrapSelection],
   282→  );
   283→
   284→  // Backlinks for this note: use the precomputed backlinks field, then
   285→  // resolve to Note objects.
   286→  const backlinks = useMemo(() => {
   287→    return note.backlinks
   288→      .map((id) => allNotes.find((n) => n.id === id))
   289→      .filter((n): n is Note => Boolean(n));
   290→  }, [allNotes, note.backlinks]);
   291→
   292→  return (
   293→    <div
   294→      style={{
   295→        flex: 1,
   296→        minWidth: 0,
   297→        background: 'var(--bg)',
   298→        display: 'flex',
   299→        flexDirection: 'column',
   300→        overflow: 'hidden',
   301→      }}
   302→    >
   303→      {/* Mode switcher tabs + formatting toolbar (toolbar only in edit mode) */}
   304→      <div style={{
   305→        display: 'flex',
   306→        alignItems: 'center',
   307→        borderBottom: '1px solid var(--bd)',
   308→        background: 'var(--bg1)',
   309→        flexShrink: 0,
   310→      }}>
   311→        <div style={{ display: 'flex', padding: '0 4px' }}>
   312→          <ModeTab icon={Pencil} label="edit" active={editor.viewMode === 'edit'} onClick={() => editor.switchViewMode('edit')} />
   313→          <ModeTab icon={Eye} label="preview" active={editor.viewMode === 'preview'} onClick={() => editor.switchViewMode('preview')} />
   314→          <ModeTab icon={GitCompare} label="diff" active={editor.viewMode === 'diff'} onClick={() => editor.switchViewMode('diff')} />
   315→        </div>
   316→        {editor.viewMode === 'edit' && (
   317→          <div style={{ flex: 1, minWidth: 0, overflow: 'hidden' }}>
   318→            <FormattingToolbar
   319→              textareaRef={textareaRef}
   320→              body={editor.body}
   321→              onBodyChange={editor.updateBody}
   322→            />
   323→          </div>
   324→        )}
   325→        {editor.viewMode !== 'edit' && (
   326→          <div style={{ flex: 1 }} />
   327→        )}
   328→      </div>
   329→
   330→      <div
   331→        className="sb-scroll"
   332→        style={{
   333→          flex: 1,
   334→          overflowY: 'auto',
   335→          padding: '28px 48px 80px',
   336→        }}
   337→      >
   338→        {/* Note header: tags + metadata */}
   339→        <div
   340→          style={{
   341→            display: 'flex',
   342→            alignItems: 'center',
   343→            gap: 8,
   344→            marginBottom: 14,
   345→            flexWrap: 'wrap',
   346→          }}
   347→        >
   348→          {note.tags.map((t) => {
   349→            const c = TAG_COLORS[t] || TAG_COLORS.strategy;
   350→            return (
   351→              <span
   352→                key={t}
   353→                style={{
   354→                  fontSize: 11,
   355→                  color: c.color,
   356→                  background: c.bg,
   357→                  border: `1px solid ${c.border}`,
   358→                  padding: '3px 8px',
   359→                  borderRadius: 3,
   360→                  fontFamily: 'inherit',
   361→                }}
   362→              >
   363→                #{t}
   364→              </span>
   365→            );
   366→          })}
   367→          <span style={{ color: 'var(--t3)', fontSize: 11 }}>·</span>
   368→          <span style={{ color: 'var(--t3)', fontSize: 11 }}>{formatDate(note.updatedAt)}</span>
   369→          <span style={{ color: 'var(--t3)', fontSize: 11 }}>·</span>
   370→          <span style={{ color: 'var(--t3)', fontSize: 11 }}>{countWords(note.body)} words</span>
   371→          <span style={{ color: 'var(--t3)', fontSize: 11 }}>·</span>
   372→          <span style={{ color: 'var(--t3)', fontSize: 11 }}>{backlinks.length} backlinks</span>
   373→          <span style={{ color: 'var(--t3)', fontSize: 11 }}>·</span>
   374→          <button
   375→            onClick={() => onToggleEvergreen(note.id)}
   376→            title="Toggle evergreen status"
   377→            style={{
   378→              fontSize: 11,
   379→              color: note.status === 'evergreen' ? 'var(--grn)' : 'var(--t3)',
   380→              background: note.status === 'evergreen' ? 'var(--grn-bg)' : 'transparent',
   381→              border:
   382→                note.status === 'evergreen' ? '1px solid #1a4a2a' : '1px solid var(--bd2)',
   383→              padding: '3px 8px',
   384→              borderRadius: 3,
   385→              cursor: 'pointer',
   386→              display: 'inline-flex',
   387→              alignItems: 'center',
   388→              gap: 4,
   389→              fontFamily: 'inherit',
   390→            }}
   391→          >
   392→            {note.status === 'evergreen' ? (
   393→              <>
   394→                <span style={{ width: 6, height: 6, borderRadius: '50%', background: 'var(--grn)' }} />
   395→                evergreen
   396→              </>
   397→            ) : (
   398→              <>
   399→                <RefreshCw size={11} />
   400→                draft
   401→              </>
   402→            )}
   403→          </button>
   404→        </div>
   405→
   406→        {/* Title */}
   407→        <input
   408→          value={editor.title}
   409→          onChange={(e) => editor.updateTitle(e.target.value)}
   410→          placeholder="Untitled note"
   411→          style={{
   412→            width: '100%',
   413→            background: 'transparent',
   414→            border: 'none',
   415→            outline: 'none',
   416→            color: 'var(--t1)',
   417→            fontSize: 28,
   418→            fontWeight: 700,
   419→            letterSpacing: '-0.02em',
   420→            padding: 0,
   421→            marginBottom: 6,
   422→            fontFamily: 'inherit',
   423→            caretColor: 'var(--acc2)',
   424→          }}
   425→        />
   426→
   427→        {/* Subtitle */}
   428→        <input
   429→          value={editor.subtitle}
   430→          onChange={(e) => editor.updateSubtitle(e.target.value)}
   431→          placeholder="A short tagline…"
   432→          style={{
   433→            width: '100%',
   434→            background: 'transparent',
   435→            border: 'none',
   436→            outline: 'none',
   437→            color: 'var(--t3)',
   438→            fontSize: 15,
   439→            fontStyle: 'italic',
   440→            padding: 0,
   441→            marginBottom: 28,
   442→            fontFamily: 'inherit',
   443→            caretColor: 'var(--acc2)',
   444→          }}
   445→        />
   446→
   447→        {/* Body: conditional on view mode */}
   448→        {editor.viewMode === 'edit' && (
   449→          <div style={{ minHeight: 200 }}>
   450→            {segments.map((seg, idx) => {
   451→              if (seg.type === 'embed' && seg.block) {
   452→                return <div key={idx}>{renderEmbed(seg.block)}</div>;
   453→              }
   454→              // Check if there are any embed segments
   455→              const hasEmbeds = segments.some((s) => s.type === 'embed');
   456→              if (!hasEmbeds) {
   457→                // No embeds — use the single overlay editor
   458→                return (
   459→                  <div
   460→                    key={idx}
   461→                    className="sb-editor-wrap"
   462→                    onClick={handleWrapClick}
   463→                    style={{ position: 'relative', minHeight: 200 }}
   464→                  >
   465→                    <pre
   466→                      ref={preRef}
   467→                      className="sb-editor-pre"
   468→                      dangerouslySetInnerHTML={{ __html: overlayHtml }}
   469→                      aria-hidden="true"
   470→                    />
   471→                    <textarea
   472→                      ref={textareaRef}
   473→                      className="sb-editor-textarea"
   474→                      value={editor.body}
   475→                      onChange={(e) => editor.updateBody(e.target.value)}
   476→                      onKeyDown={(e) => { handleKeyDown(e); handleSlashDetect(e); }}
   477→                      placeholder="# Start writing your note…  ·  type / to insert embed"
   478→                      spellCheck={false}
   479→                      rows={1}
   480→                    />
   481→                  </div>
   482→                );
   483→              }
   484→              // Multiple segments — render text segments as editable textareas with overlay
   485→              // Each text segment gets its own mini editor that edits that portion of the body
   486→              const segStart = seg.startOffset || 0;
   487→              const segEnd = seg.endOffset || 0;
   488→              const segContent = seg.content || '';
   489→              return (
   490→                <SegmentEditor
   491→                  key={idx}
   492→                  content={segContent}
   493→                  startOffset={segStart}
   494→                  fullBody={editor.body}
   495→                  onBodyChange={editor.updateBody}
   496→                  onWrapClick={handleWrapClick}
   497→                  onKeyDown={handleSlashDetect}
   498→                />
   499→              );
   500→            })}
   501→            {segments.length > 1 && (
   502→              <div style={{
   503→                marginTop: 12,
   504→                padding: '8px 12px',
   505→                background: 'var(--bg1)',
   506→                border: '1px solid var(--bd)',
   507→                borderRadius: 4,
   508→                fontSize: 11,
   509→                color: 'var(--t3)',
   510→                fontStyle: 'italic',
   511→              }}>
   512→                this note contains embed blocks. text is editable above — embeds are interactive (drag, edit, expand).
   513→              </div>
   514→            )}
   515→          </div>
   516→        )}
   517→
   518→        {editor.viewMode === 'preview' && (
   519→          <div style={{ minHeight: 200 }}>
   520→            {segments.map((seg, idx) => {
   521→              if (seg.type === 'embed' && seg.block) {
   522→                return <div key={idx}>{renderEmbed(seg.block)}</div>;
   523→              }
   524→              return (
   525→                <div
   526→                  key={idx}
   527→                  className="sb-preview"
   528→                  dangerouslySetInnerHTML={{ __html: renderMarkdownHtml(seg.content || '') }}
   529→                  onClick={(e) => {
   530→                    if (!e.metaKey && !e.ctrlKey) return;
   531→                    const target = e.target as HTMLElement;
   532→                    const wikiEl = target.closest('[data-wiki]') as HTMLElement | null;
   533→                    if (wikiEl) {
   534→                      const title = wikiEl.getAttribute('data-wiki');
   535→                      if (title) {
   536→                        e.preventDefault();
   537→                        onOpenNoteByTitle(title);
   538→                      }
   539→                    }
   540→                  }}
   541→                  style={{ cursor: 'default' }}
   542→                />
   543→              );
   544→            })}
   545→          </div>
   546→        )}
   547→
   548→        {editor.viewMode === 'diff' && (
   549→          <DiffView oldText={editor.savedBody} newText={editor.body} />
   550→        )}
   551→
   552→        {/* Backlinks section */}
   553→        <div
   554→          style={{
   555→            marginTop: 40,
   556→            background: 'var(--bg2)',
   557→            border: '1px solid var(--bd)',
   558→            borderRadius: 4,
   559→            padding: '12px 16px',
   560→          }}
   561→        >
   562→          <div
   563→            style={{
   564→              fontSize: 11,
   565→              textTransform: 'uppercase',
   566→              letterSpacing: '0.09em',
   567→              color: 'var(--t3)',
   568→              marginBottom: 10,
   569→              fontWeight: 600,
   570→            }}
   571→          >
   572→            backlinks — {backlinks.length} {backlinks.length === 1 ? 'note' : 'notes'} reference this
   573→          </div>
   574→          {backlinks.length === 0 ? (
   575→            <div style={{ fontSize: 13, color: 'var(--t3)', fontStyle: 'italic' }}>
   576→              no backlinks yet. link this note from elsewhere with [[{note.title}]].
   577→            </div>
   578→          ) : (
   579→            <div style={{ display: 'flex', flexDirection: 'column', gap: 6 }}>
   580→              {backlinks.map((b) => (
   581→                <button
   582→                  key={b.id}

   685→/* ---------- Diff View (GitHub-style) ---------- */
   686→
   687→function DiffView({ oldText, newText }: { oldText: string; newText: string }) {
   688→  const diff = useMemo(() => computeDiff(oldText, newText), [oldText, newText]);
   689→
   690→  if (!diff.hasChanges) {
   691→    return (
   692→      <div className="sb-diff">
   693→        <div className="sb-diff-empty">
   694→          no unsaved changes — everything is in sync
   695→        </div>
   696→      </div>
   697→    );
   698→  }
   699→
   700→  return (
   701→    <div className="sb-diff">
   702→      <div className="sb-diff-header">
   703→        <span>comparing saved version → working copy</span>
   704→        <div className="sb-diff-stat">
   705→          <span className="sb-diff-add-count">+{diff.addedCount}</span>
   706→          <span className="sb-diff-del-count">−{diff.removedCount}</span>
   707→        </div>
   708→      </div>
   709→      <div className="sb-diff-body">
   710→        {diff.lines.map((line, i) => (
   711→          <div key={i} className={`sb-diff-line ${line.type}`}>
   712→            <span className="sb-diff-line-num">
   713→              {line.type === 'unchanged' && line.oldLineNum}
   714→              {line.type === 'removed' && line.oldLineNum}
   715→              {line.type === 'added' && line.newLineNum}
   716→            </span>
   717→            <span className="sb-diff-line-num" style={{ borderLeft: '1px solid var(--bd)', paddingLeft: 4, width: 38 }}>
   718→              {line.type === 'unchanged' && line.newLineNum}
   719→              {line.type === 'removed' && ''}
   720→              {line.type === 'added' && line.newLineNum}
   721→            </span>
   722→            <span className="sb-diff-line-sign">
   723→              {line.type === 'added' ? '+' : line.type === 'removed' ? '−' : ' '}
   724→            </span>
   725→            <span className="sb-diff-line-content">
   726→              {line.content || ' '}
   727→            </span>
   728→          </div>
   729→        ))}
   730→      </div>
   731→    </div>
   732→  );
   733→}

   735→/* ---------- Segment Editor (for notes with embeds) ---------- */
   736→/* Renders a text segment as an editable textarea + overlay, splicing
   737→   changes back into the full body at the correct offset. */
   738→
   739→function SegmentEditor({
   740→  content,
   741→  startOffset,
   742→  fullBody,
   743→  onBodyChange,
   744→  onWrapClick,
   745→  onKeyDown,
   746→}: {
   747→  content: string;
   748→  startOffset: number;
   749→  fullBody: string;
   750→  onBodyChange: (next: string) => void;
   751→  onWrapClick: (e: React.MouseEvent) => void;
   752→  onKeyDown: (e: React.KeyboardEvent<HTMLTextAreaElement>) => void;
   753→}) {
   754→  const segTextareaRef = useRef<HTMLTextAreaElement>(null);
   755→  const segPreRef = useRef<HTMLPreElement>(null);
   756→  const segOverlayHtml = useMemo(() => renderMarkdownOverlay(content), [content]);
   757→
   758→  const autoResize = useCallback(() => {
   759→    const ta = segTextareaRef.current;
   760→    if (!ta) return;
   761→    ta.style.height = 'auto';
   762→    ta.style.height = ta.scrollHeight + 'px';
   763→  }, []);
   764→
   765→  useEffect(() => { autoResize(); }, [content, autoResize]);
   766→
   767→  const handleChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => {
   768→    const newSegContent = e.target.value;
   769→    // Splice the new content into the full body at startOffset
   770→    const before = fullBody.slice(0, startOffset);
   771→    const after = fullBody.slice(startOffset + content.length);
   772→    onBodyChange(before + newSegContent + after);
   773→  }, [fullBody, startOffset, content.length, onBodyChange]);
   774→
   775→  return (
   776→    <div
   777→      className="sb-editor-wrap"
   778→      onClick={onWrapClick}
   779→      style={{ position: 'relative', minHeight: content ? undefined : 60 }}
   780→    >
   781→      <pre
   782→        ref={segPreRef}
   783→        className="sb-editor-pre"
   784→        dangerouslySetInnerHTML={{ __html: segOverlayHtml }}
   785→        aria-hidden="true"
   786→      />
   787→      <textarea
   788→        ref={segTextareaRef}
   789→        className="sb-editor-textarea"
   790→        value={content}
   791→        onChange={handleChange}
   792→        onKeyDown={onKeyDown}
   793→        placeholder={content ? "" : "click here to start writing…"}
   794→        spellCheck={false}
   795→        rows={1}
   796→        style={{ height: 'auto', minHeight: content ? undefined : 60 }}
   797→      />
   798→    </div>
   799→  );
   800→}
