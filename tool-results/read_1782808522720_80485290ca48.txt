     1→     1→'use client';
     2→     2→
     3→     3→import { useMemo, useState, useRef, useEffect, useCallback } from 'react';
     4→     4→import { Note, Folder, TAG_COLORS } from '@/types';
     5→     5→import { forceDirectedLayout } from '@/utils/graph';
     6→     6→import { countWords } from '@/utils/markdown';
     7→     7→import {
     8→     8→  Search, X, Tag, Folder as FolderIcon, Hash, ArrowRight, Maximize2,
     9→     9→  Network, TrendingUp,
    10→    10→} from 'lucide-react';
    11→    11→
    12→    12→interface GraphViewProps {
    13→    13→  notes: Note[];
    14→    14→  folders: Folder[];
    15→    15→  onOpenNote: (id: string) => void;
    16→    16→  onBack: () => void;
    17→    17→}
    18→    18→
    19→    19→interface FilterState {
    20→    20→  search: string;
    21→    21→  tag: string | null;
    22→    22→  folderId: string | null;
    23→    23→  showLabels: boolean;
    24→    24→  showIsolated: boolean;
    25→    25→}
    26→    26→
    27→    27→export function GraphView({ notes, folders, onOpenNote, onBack }: GraphViewProps) {
    28→    28→  const [filter, setFilter] = useState<FilterState>({
    29→    29→    search: '',
    30→    30→    tag: null,
    31→    31→    folderId: null,
    32→    32→    showLabels: true,
    33→    33→    showIsolated: true,
    34→    34→  });
    35→    35→  const [hoveredNode, setHoveredNode] = useState<string | null>(null);
    36→    36→  const [selectedNode, setSelectedNode] = useState<string | null>(null);
    37→    37→  const svgRef = useRef<SVGSVGElement>(null);
    38→    38→  const [svgDims, setSvgDims] = useState({ width: 800, height: 600 });
    39→    39→
    40→    40→  // Measure available space
    41→    41→  useEffect(() => {
    42→    42→    const measure = () => {
    43→    43→      const el = svgRef.current?.parentElement;
    44→    44→      if (el) {
    45→    45→        setSvgDims({
    46→    46→          width: el.clientWidth,
    47→    47→          height: el.clientHeight,
    48→    48→        });
    49→    49→      }
    50→    50→    };
    51→    51→    measure();
    52→    52→    window.addEventListener('resize', measure);
    53→    53→    return () => window.removeEventListener('resize', measure);
    54→    54→  }, []);
    55→    55→
    56→    56→  // Build edges from wiki-links (deduplicated)
    57→    57→  const allEdges = useMemo(() => {
    58→    58→    const titleToId = new Map<string, string>();
    59→    59→    for (const n of notes) {
    60→    60→      titleToId.set(n.title.toLowerCase(), n.id);
    61→    61→      titleToId.set(n.filename.toLowerCase().replace(/\.md$/, ''), n.id);
    62→    62→    }
    63→    63→    const edges: { from: string; to: string }[] = [];
    64→    64→    const seen = new Set<string>();
    65→    65→    for (const n of notes) {
    66→    66→      const re = /\[\[([^\]]+)\]\]/g;
    67→    67→      let m: RegExpExecArray | null;
    68→    68→      while ((m = re.exec(n.body)) !== null) {
    69→    69→        const targetId = titleToId.get(m[1].toLowerCase());
    70→    70→        if (targetId && targetId !== n.id) {
    71→    71→          const key = [n.id, targetId].sort().join('→');
    72→    72→          if (!seen.has(key)) {
    73→    73→            seen.add(key);
    74→    74→            edges.push({ from: n.id, to: targetId });
    75→    75→          }
    76→    76→        }
    77→    77→      }
    78→    78→    }
    79→    79→    return edges;
    80→    80→  }, [notes]);
    81→    81→
    82→    82→  // Apply filters
    83→    83→  const filteredNotes = useMemo(() => {
    84→    84→    return notes.filter((n) => {
    85→    85→      if (filter.tag !== null) {
    86→    86→        if (filter.tag === 'untagged' && n.tags.length > 0) return false;
    87→    87→        if (filter.tag !== 'untagged' && !n.tags.includes(filter.tag)) return false;
    88→    88→      }
    89→    89→      if (filter.folderId !== null) {
    90→    90→        // match folder or any descendant
    91→    91→        const isDescendant = (fid: string): boolean => {
    92→    92→          if (fid === filter.folderId) return true;
    93→    93→          let cur = folders.find((f) => f.id === fid)?.parentId;
    94→    94→          while (cur) {
    95→    95→            if (cur === filter.folderId) return true;
    96→    96→            cur = folders.find((f) => f.id === cur)?.parentId;
    97→    97→          }
    98→    98→          return false;
    99→    99→        };
   100→   100→        if (!isDescendant(n.folderId)) return false;
   101→   101→      }
   102→   102→      if (filter.search.trim()) {
   103→   103→        const q = filter.search.toLowerCase();
   104→   104→        if (!n.title.toLowerCase().includes(q) && !n.filename.toLowerCase().includes(q)) return false;
   105→   105→      }
   106→   106→      return true;
   107→   107→    });
   108→   108→  }, [notes, filter, folders]);
   109→   109→
   110→   110→  const filteredNoteIds = useMemo(() => new Set(filteredNotes.map((n) => n.id)), [filteredNotes]);
   111→   111→
   112→   112→  // Edges that connect two filtered notes
   113→   113→  const filteredEdges = useMemo(() => {
   114→   114→    if (!filter.showIsolated) {
   115→   115→      // only show notes that have at least one connection in the filtered set
   116→   116→    }
   117→   117→    return allEdges.filter((e) => filteredNoteIds.has(e.from) && filteredNoteIds.has(e.to));
   118→   118→  }, [allEdges, filteredNoteIds, filter.showIsolated]);
   119→   119→
   120→   120→  // Filter out isolated nodes if toggle is off
   121→   121→  const visibleNotes = useMemo(() => {
   122→   122→    if (filter.showIsolated) return filteredNotes;
   123→   123→    const connectedIds = new Set<string>();
   124→   124→    for (const e of filteredEdges) {
   125→   125→      connectedIds.add(e.from);
   126→   126→      connectedIds.add(e.to);
   127→   127→    }
   128→   128→    return filteredNotes.filter((n) => connectedIds.has(n.id));
   129→   129→  }, [filteredNotes, filteredEdges, filter.showIsolated]);
   130→   130→
   131→   131→  // Compute layout
   132→   132→  const layout = useMemo(() => {
   133→   133→    return forceDirectedLayout(
   134→   134→      visibleNotes.map((n) => n.id),
   135→   135→      filteredEdges,
   136→   136→      selectedNode,
   137→   137→      svgDims.width,
   138→   138→      svgDims.height - 20,
   139→   139→      500,
   140→   140→    );
   141→   141→  }, [visibleNotes, filteredEdges, selectedNode, svgDims]);
   142→   142→
   143→   143→  const nodeMap = useMemo(() => new Map(layout.nodes.map((n) => [n.id, n])), [layout]);
   144→   144→  const noteById = useMemo(() => new Map(notes.map((n) => [n.id, n])), [notes]);
   145→   145→
   146→   146→  // Compute stats
   147→   147→  const stats = useMemo(() => {
   148→   148→    const nodeCount = visibleNotes.length;
   149→   149→    const edgeCount = filteredEdges.length;
   150→   150→    const maxPossibleEdges = (nodeCount * (nodeCount - 1)) / 2;
   151→   151→    const density = maxPossibleEdges > 0 ? (edgeCount / maxPossibleEdges) * 100 : 0;
   152→   152→    // degree per note
   153→   153→    const degreeMap = new Map<string, number>();
   154→   154→    for (const e of filteredEdges) {
   155→   155→      degreeMap.set(e.from, (degreeMap.get(e.from) || 0) + 1);
   156→   156→      degreeMap.set(e.to, (degreeMap.get(e.to) || 0) + 1);
   157→   157→    }
   158→   158→    const hubs = visibleNotes
   159→   159→      .map((n) => ({ note: n, degree: degreeMap.get(n.id) || 0 }))
   160→   160→      .sort((a, b) => b.degree - a.degree)
   161→   161→      .slice(0, 8);
   162→   162→    const isolated = visibleNotes.filter((n) => !degreeMap.has(n.id)).length;
   163→   163→    return { nodeCount, edgeCount, density, hubs, isolated };
   164→   164→  }, [visibleNotes, filteredEdges]);
   165→   165→
   166→   166→  // All tags for filter
   167→   167→  const allTags = useMemo(() => {
   168→   168→    const m = new Map<string, number>();
   169→   169→    for (const n of notes) {
   170→   170→      if (n.tags.length === 0) m.set('untagged', (m.get('untagged') || 0) + 1);
   171→   171→      else for (const t of n.tags) m.set(t, (m.get(t) || 0) + 1);
   172→   172→    }
   173→   173→    return Array.from(m.entries()).sort((a, b) => b[1] - a[1]);
   174→   174→  }, [notes]);
   175→   175→
   176→   176→  // Top-level folders for filter
   177→   177→  const topFolders = useMemo(() => folders.filter((f) => f.parentId === null), [folders]);
   178→   178→
   179→   179→  // Helpers
   180→   180→  const getNodeColor = useCallback((noteId: string): string => {
   181→   181→    const note = noteById.get(noteId);
   182→   182→    if (!note) return '#3d378a';
   183→   183→    if (note.tags.length > 0) {
   184→   184→      const tag = note.tags[0];
   185→   185→      const c = TAG_COLORS[tag];
   186→   186→      if (c) return c.border;
   187→   187→    }
   188→   188→    return '#534AB7';
   189→   189→  }, [noteById]);
   190→   190→
   191→   191→  const getNodeFill = useCallback((noteId: string): string => {
   192→   192→    const note = noteById.get(noteId);
   193→   193→    if (!note) return '#3d378a';
   194→   194→    if (note.tags.length > 0) {
   195→   195→      const tag = note.tags[0];
   196→   196→      const c = TAG_COLORS[tag];
   197→   197→      if (c) return c.color;
   198→   198→    }
   199→   199→    return '#534AB7';
   200→   200→  }, [noteById]);
   201→   201→
   202→   202→  return (
   203→   203→    <div style={{ flex: 1, display: 'flex', overflow: 'hidden', background: 'var(--bg)' }}>
   204→   204→      {/* Left sidebar: filters + stats */}
   205→   205→      <div
   206→   206→        style={{
   207→   207→          width: 260,
   208→   208→          minWidth: 260,
   209→   209→          background: 'var(--bg1)',
   210→   210→          borderRight: '1px solid var(--bd)',
   211→   211→          display: 'flex',
   212→   212→          flexDirection: 'column',
   213→   213→          overflow: 'hidden',
   214→   214→        }}
   215→   215→      >
   216→   216→        <div className="sb-scroll" style={{ flex: 1, overflowY: 'auto', padding: '16px 14px' }}>
   217→   217→          {/* Back button */}
   218→   218→          <button
   219→   219→            onClick={onBack}
   220→   220→            style={{
   221→   221→              background: 'transparent',
   222→   222→              border: '1px solid var(--bd2)',
   223→   223→              borderRadius: 4,
   224→   224→              padding: '6px 10px',
   225→   225→              color: 'var(--t2)',
   226→   226→              fontSize: 12,
   227→   227→              fontFamily: 'inherit',
   228→   228→              cursor: 'pointer',
   229→   229→              display: 'flex',
   230→   230→              alignItems: 'center',
   231→   231→              gap: 6,
   232→   232→              marginBottom: 18,
   233→   233→            }}
   234→   234→            onMouseEnter={(e) => { e.currentTarget.style.background = 'var(--bg3)'; e.currentTarget.style.color = 'var(--t1)'; }}
   235→   235→            onMouseLeave={(e) => { e.currentTarget.style.background = 'transparent'; e.currentTarget.style.color = 'var(--t2)'; }}
   236→   236→          >
   237→   237→            <ArrowRight size={12} style={{ transform: 'rotate(180deg)' }} />
   238→   238→            back
   239→   239→          </button>
   240→   240→
   241→   241→          {/* Title */}
   242→   242→          <div style={{ marginBottom: 18 }}>
   243→   243→            <div style={{ display: 'flex', alignItems: 'center', gap: 8, marginBottom: 4 }}>
   244→   244→              <Network size={16} color="var(--acc2)" />
   245→   245→              <h1 style={{ fontSize: 18, fontWeight: 700, color: 'var(--t1)', margin: 0 }}>
   246→   246→                knowledge graph
   247→   247→              </h1>
   248→   248→            </div>
   249→   249→            <p style={{ fontSize: 12, color: 'var(--t3)', margin: 0 }}>
   250→   250→              visualize how your notes connect
   251→   251→            </p>
   252→   252→          </div>
   253→   253→
   254→   254→          {/* Search */}
   255→   255→          <div style={{ marginBottom: 16 }}>
   256→   256→            <Label>search</Label>
   257→   257→            <div style={{
   258→   258→              display: 'flex',
   259→   259→              alignItems: 'center',
   260→   260→              gap: 8,
   261→   261→              background: 'var(--bg3)',
   262→   262→              border: '1px solid var(--bd2)',
   263→   263→              borderRadius: 4,
   264→   264→              padding: '0 10px',
   265→   265→              height: 32,
   266→   266→            }}>
   267→   267→              <Search size={13} color="var(--t3)" />
   268→   268→              <input
   269→   269→                value={filter.search}
   270→   270→                onChange={(e) => setFilter((f) => ({ ...f, search: e.target.value }))}
   271→   271→                placeholder="filter by title…"
   272→   272→                style={{
   273→   273→                  flex: 1,
   274→   274→                  background: 'transparent',
   275→   275→                  border: 'none',
   276→   276→                  outline: 'none',
   277→   277→                  color: 'var(--t1)',
   278→   278→                  fontSize: 12,
   279→   279→                  fontFamily: 'inherit',
   280→   280→                  caretColor: 'var(--acc2)',
   281→   281→                }}
   282→   282→              />
   283→   283→              {filter.search && (
   284→   284→                <button
   285→   285→                  onClick={() => setFilter((f) => ({ ...f, search: '' }))}
   286→   286→                  style={{ background: 'transparent', border: 'none', color: 'var(--t3)', cursor: 'pointer', padding: 0 }}
   287→   287→                >
   288→   288→                  <X size={12} />
   289→   289→                </button>
   290→   290→              )}
   291→   291→            </div>
   292→   292→          </div>
   293→   293→
   294→   294→          {/* Tag filter */}
   295→   295→          <div style={{ marginBottom: 16 }}>
   296→   296→            <Label>filter by tag</Label>
   297→   297→            <div style={{ display: 'flex', flexDirection: 'column', gap: 3 }}>
   298→   298→              <FilterChip
   299→   299→                active={filter.tag === null}
   300→   300→                onClick={() => setFilter((f) => ({ ...f, tag: null }))}
   301→   301→                label="all tags"
   302→   302→                count={notes.length}
   303→   303→              />
   304→   304→              {allTags.map(([tag, count]) => (
   305→   305→                <FilterChip
   306→   306→                  key={tag}
   307→   307→                  active={filter.tag === tag}
   308→   308→                  onClick={() => setFilter((f) => ({ ...f, tag: f.tag === tag ? null : tag }))}
   309→   309→                  label={tag === 'untagged' ? 'untagged' : `#${tag}`}
   310→   310→                  count={count}
   311→   311→                  color={tag !== 'untagged' ? TAG_COLORS[tag]?.color : undefined}
   312→   312→                />
   313→   313→              ))}
   314→   314→            </div>
   315→   315→          </div>
   316→   316→
   317→   317→          {/* Folder filter */}
   318→   318→          <div style={{ marginBottom: 16 }}>
   319→   319→            <Label>filter by folder</Label>
   320→   320→            <div style={{ display: 'flex', flexDirection: 'column', gap: 3 }}>
   321→   321→              <FilterChip
   322→   322→                active={filter.folderId === null}
   323→   323→                onClick={() => setFilter((f) => ({ ...f, folderId: null }))}
   324→   324→                label="all folders"
   325→   325→                count={notes.length}
   326→   326→              />
   327→   327→              {topFolders.map((folder) => {
   328→   328→                const count = notes.filter((n) => {
   329→   329→                  let cur = n.folderId;
   330→   330→                  while (cur) {
   331→   331→                    if (cur === folder.id) return true;
   332→   332→                    cur = folders.find((f) => f.id === cur)?.parentId || null;
   333→   333→                  }
   334→   334→                  return false;
   335→   335→                }).length;
   336→   336→                return (
   337→   337→                  <FilterChip
   338→   338→                    key={folder.id}
   339→   339→                    active={filter.folderId === folder.id}
   340→   340→                    onClick={() => setFilter((f) => ({ ...f, folderId: f.folderId === folder.id ? null : folder.id }))}
   341→   341→                    label={folder.name}
   342→   342→                    count={count}
   343→   343→                  />
   344→   344→                );
   345→   345→              })}
   346→   346→            </div>
   347→   347→          </div>
   348→   348→
   349→   349→          {/* Toggles */}
   350→   350→          <div style={{ marginBottom: 18 }}>
   351→   351→            <Label>display</Label>
   352→   352→            <Toggle
   353→   353→              checked={filter.showLabels}
   354→   354→              onChange={(v) => setFilter((f) => ({ ...f, showLabels: v }))}
   355→   355→              label="show labels"
   356→   356→            />
   357→   357→            <Toggle
   358→   358→              checked={filter.showIsolated}
   359→   359→              onChange={(v) => setFilter((f) => ({ ...f, showIsolated: v }))}
   360→   360→              label="show isolated notes"
   361→   361→            />
   362→   362→          </div>
   363→   363→
   364→   364→          {/* Stats */}
   365→   365→          <div style={{
   366→   366→            background: 'var(--bg2)',
   367→   367→            border: '1px solid var(--bd)',
   368→   368→            borderRadius: 5,
   369→   369→            padding: '12px 14px',
   370→   370→            marginBottom: 16,
   371→   371→          }}>
   372→   372→            <Label>graph stats</Label>
   373→   373→            <StatRow label="nodes" value={stats.nodeCount} />
   374→   374→            <StatRow label="edges" value={stats.edgeCount} />
   375→   375→            <StatRow label="density" value={`${stats.density.toFixed(1)}%`} />
   376→   376→            <StatRow label="isolated" value={stats.isolated} />
   377→   377→            <StatRow label="avg degree" value={stats.nodeCount > 0 ? (stats.edgeCount * 2 / stats.nodeCount).toFixed(1) : '0'} />
   378→   378→          </div>
   379→   379→
   380→   380→          {/* Hub notes */}
   381→   381→          {stats.hubs.length > 0 && (
   382→   382→            <div>
   383→   383→              <Label>hub notes (most connected)</Label>
   384→   384→              <div style={{ display: 'flex', flexDirection: 'column', gap: 2 }}>
   385→   385→                {stats.hubs.map(({ note, degree }) => (
   386→   386→                  <button
   387→   387→                    key={note.id}
   388→   388→                    onClick={() => {
   389→   389→                      setSelectedNode(note.id);
   390→   390→                      onOpenNote(note.id);
   391→   391→                    }}
   392→   392→                    style={{
   393→   393→                      background: 'transparent',
   394→   394→                      border: 'none',
   395→   395→                      borderBottom: '1px solid var(--bd)',
   396→   396→                      borderRadius: 0,
   397→   397→                      padding: '7px 0',
   398→   398→                      cursor: 'pointer',
   399→   399→                      textAlign: 'left',
   400→   400→                      fontFamily: 'inherit',
   401→   401→                      display: 'flex',
   402→   402→                      alignItems: 'center',
   403→   403→                      gap: 8,
   404→   404→                    }}
   405→   405→                    onMouseEnter={(e) => { e.currentTarget.style.background = 'var(--bg2)'; }}
   406→   406→                    onMouseLeave={(e) => { e.currentTarget.style.background = 'transparent'; }}
   407→   407→                  >
   408→   408→                    <span style={{
   409→   409→                      fontSize: 11,
   410→   410→                      color: 'var(--acc2)',
   411→   411→                      fontWeight: 600,
   412→   412→                      minWidth: 16,
   413→   413→                    }}>
   414→   414→                      {degree}
   415→   415→                    </span>
   416→   416→                    <span style={{
   417→   417→                      fontSize: 12,
   418→   418→                      color: 'var(--t2)',
   419→   419→                      flex: 1,
   420→   420→                      overflow: 'hidden',
   421→   421→                      textOverflow: 'ellipsis',
   422→   422→                      whiteSpace: 'nowrap',
   423→   423→                    }}>
   424→   424→                      {note.title}
   425→   425→                    </span>
   426→   426→                  </button>
   427→   427→                ))}
   428→   428→              </div>
   429→   429→            </div>
   430→   430→          )}
   431→   431→        </div>
   432→   432→      </div>
   433→   433→
   434→   434→      {/* Main canvas: the graph */}
   435→   435→      <div style={{ flex: 1, position: 'relative', overflow: 'hidden', background: 'var(--bg)' }}>
   436→   436→        {visibleNotes.length === 0 ? (
   437→   437→          <div style={{
   438→   438→            position: 'absolute',
   439→   439→            inset: 0,
   440→   440→            display: 'flex',
   441→   441→            alignItems: 'center',
   442→   442→            justifyContent: 'center',
   443→   443→            color: 'var(--t3)',
   444→   444→            fontSize: 14,
   445→   445→            fontStyle: 'italic',
   446→   446→          }}>
   447→   447→            no notes match your filters
   448→   448→          </div>
   449→   449→        ) : (
   450→   450→          <svg
   451→   451→            ref={svgRef}
   452→   452→            width="100%"
   453→   453→            height="100%"
   454→   454→            viewBox={`0 0 ${svgDims.width} ${svgDims.height}`}
   455→   455→            style={{ display: 'block' }}
   456→   456→          >
   457→   457→            {/* Edges */}
   458→   458→            {layout.edges.map((e, i) => {
   459→   459→              const a = nodeMap.get(e.from);
   460→   460→              const b = nodeMap.get(e.to);
   461→   461→              if (!a || !b) return null;
   462→   462→              const isHighlighted = hoveredNode && (e.from === hoveredNode || e.to === hoveredNode) ||
   463→   463→                                    selectedNode && (e.from === selectedNode || e.to === selectedNode);
   464→   464→              return (
   465→   465→                <line
   466→   466→                  key={i}
   467→   467→                  x1={a.x}
   468→   468→                  y1={a.y}
   469→   469→                  x2={b.x}
   470→   470→                  y2={b.y}
   471→   471→                  stroke={isHighlighted ? 'var(--acc)' : '#3d378a'}
