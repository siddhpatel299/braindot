    80→    80→export function ReadingView({
    81→    81→  libraryItems, highlights, notes, onBack, onOpenNote,
    82→    82→  onAddLibraryItem, onUpdateLibraryItem, onDeleteLibraryItem,
    83→    83→  onAddHighlight, onUpdateHighlight, onDeleteHighlight,
    84→    84→  onCreateNoteFromHighlight,
    85→    85→}: ReadingViewProps) {
    86→    86→  const [search, setSearch] = useState('');
    87→    87→  const [tabFilter, setTabFilter] = useState<TabFilter>('all');
    88→    88→  const [activeItemId, setActiveItemId] = useState<string | null>(libraryItems[0]?.id || null);
    89→    89→  const [sidebarTab, setSidebarTab] = useState<SidebarTab>('ai');
    90→    90→  const [currentHighlightColor, setCurrentHighlightColor] = useState<HighlightColor>('yellow');
    91→    91→  const [showImport, setShowImport] = useState(false);
    92→    92→  const [showSelectionToolbar, setShowSelectionToolbar] = useState(false);
    93→    93→  const [selectionPos, setSelectionPos] = useState({ x: 0, y: 0 });
    94→    94→  const [selectedText, setSelectedText] = useState('');
    95→    95→  const readerRef = useRef<HTMLDivElement>(null);
    96→    96→
    97→    97→  // Live news/papers state
    98→    98→  const [showFeedPanel, setShowFeedPanel] = useState<'news' | 'papers' | null>(null);
    99→    99→  const [newsCategory, setNewsCategory] = useState('tech');
   100→   100→  const [paperCategory, setPaperCategory] = useState('ai');
   101→   101→  const [fetchedItems, setFetchedItems] = useState<FetchedItem[]>([]);
   102→   102→  const [feedLoading, setFeedLoading] = useState(false);
   103→   103→  const [feedError, setFeedError] = useState<string | null>(null);
   104→   104→
   105→   105→  const activeItem = useMemo(
   106→   106→    () => libraryItems.find((l) => l.id === activeItemId) || null,
   107→   107→    [libraryItems, activeItemId],
   108→   108→  );
   109→   109→
   110→   110→  const itemHighlights = useMemo(
   111→   111→    () => highlights.filter((h) => h.libraryItemId === activeItemId),
   112→   112→    [highlights, activeItemId],
   113→   113→  );
   114→   114→
   115→   115→  const filteredItems = useMemo(() => {
   116→   116→    let items = libraryItems;
   117→   117→    if (tabFilter === 'books') items = items.filter((l) => l.type === 'epub');
   118→   118→    else if (tabFilter === 'papers') items = items.filter((l) => l.type === 'pdf');
   119→   119→    else if (tabFilter === 'news') items = items.filter((l) => l.type === 'rss' || l.type === 'url');
   120→   120→    if (search.trim()) {
   121→   121→      const q = search.toLowerCase();
   122→   122→      items = items.filter((l) => l.title.toLowerCase().includes(q) || (l.author?.toLowerCase().includes(q) ?? false));
   123→   123→    }
   124→   124→    return items;
   125→   125→  }, [libraryItems, tabFilter, search]);
   126→   126→
   127→   127→  const sections = useMemo(() => {
   128→   128→    const reading = filteredItems.filter((l) => l.status === 'reading');
   129→   129→    const papers = filteredItems.filter((l) => l.type === 'pdf' && l.status !== 'reading');
   130→   130→    const news = filteredItems.filter((l) => (l.type === 'rss' || l.type === 'url') && l.status !== 'reading');
   131→   131→    const books = filteredItems.filter((l) => l.type === 'epub' && l.status !== 'reading' && l.status !== 'done');
   132→   132→    const done = filteredItems.filter((l) => l.status === 'done');
   133→   133→    return { reading, papers, news, books, done };
   134→   134→  }, [filteredItems]);
   135→   135→
   136→   136→  // Render reader content with highlights applied
   137→   137→  const readerHtml = useMemo(() => {
   138→   138→    if (!activeItem?.content) return '';
   139→   139→    let html = renderMarkdownHtml(activeItem.content);
   140→   140→    // Apply highlights by wrapping matching text
   141→   141→    for (const hl of itemHighlights) {
   142→   142→      const escapedText = hl.text.replace(/[&<>"']/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c] || c));
   143→   143→      const re = new RegExp(escapedText.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), 'g');
   144→   144→      html = html.replace(re, `<mark class="hl-${hl.color}" data-hl-id="${hl.id}">${escapedText}</mark>`);
   145→   145→    }
   146→   146→    return html;
   147→   147→  }, [activeItem, itemHighlights]);
   148→   148→
   149→   149→  // Handle text selection in reader
   150→   150→  useEffect(() => {
   151→   151→    const handleMouseUp = (e: MouseEvent) => {
   152→   152→      const sel = window.getSelection();
   153→   153→      if (!sel || sel.isCollapsed || sel.toString().trim().length < 3) {
   154→   154→        setShowSelectionToolbar(false);
   155→   155→        return;
   156→   156→      }
   157→   157→      const range = sel.getRangeAt(0);
   158→   158→      const rect = range.getBoundingClientRect();
   159→   159→      const readerEl = readerRef.current;
   160→   160→      if (!readerEl) return;
   161→   161→      const readerRect = readerEl.getBoundingClientRect();
   162→   162→      if (rect.top < readerRect.top || rect.bottom > readerRect.bottom) {
   163→   163→        setShowSelectionToolbar(false);
   164→   164→        return;
   165→   165→      }
   166→   166→      setSelectedText(sel.toString().trim());
   167→   167→      setSelectionPos({ x: rect.left + rect.width / 2, y: rect.top - 8 });
   168→   168→      setShowSelectionToolbar(true);
   169→   169→    };
   170→   170→    document.addEventListener('mouseup', handleMouseUp);
   171→   171→    return () => document.removeEventListener('mouseup', handleMouseUp);
   172→   172→  }, []);
   173→   173→
   174→   174→  // Create a real highlight
   175→   175→  const handleHighlight = useCallback(() => {
   176→   176→    if (!selectedText || !activeItemId) return;
   177→   177→    onAddHighlight({
   178→   178→      libraryItemId: activeItemId,
   179→   179→      text: selectedText,
   180→   180→      color: currentHighlightColor,
   181→   181→      location: 'selection',
   182→   182→    });
   183→   183→    setSelectedText('');
   184→   184→    window.getSelection()?.removeAllRanges();
   185→   185→    setShowSelectionToolbar(false);
   186→   186→    setSidebarTab('highlights');
   187→   187→  }, [selectedText, activeItemId, currentHighlightColor, onAddHighlight]);
   188→   188→
   189→   189→  // Fetch news from API
   190→   190→  const fetchNews = useCallback(async (cat: string) => {
   191→   191→    setFeedLoading(true);
   192→   192→    setFeedError(null);
   193→   193→    try {
   194→   194→      const res = await fetch(`/api/reading/news?category=${cat}&XTransformPort=3000`);
   195→   195→      if (!res.ok) throw new Error('Failed to fetch news');
   196→   196→      const data = await res.json();
   197→   197→      setFetchedItems(data.items || []);
   198→   198→    } catch (e) {
   199→   199→      setFeedError(e instanceof Error ? e.message : 'Failed to fetch');
   200→   200→      setFetchedItems([]);
   201→   201→    } finally {
   202→   202→      setFeedLoading(false);
   203→   203→    }
   204→   204→  }, []);
   205→   205→
   206→   206→  // Fetch papers from API
   207→   207→  const fetchPapers = useCallback(async (cat: string) => {
   208→   208→    setFeedLoading(true);
   209→   209→    setFeedError(null);
   210→   210→    try {
   211→   211→      const res = await fetch(`/api/reading/papers?category=${cat}&XTransformPort=3000`);
   212→   212→      if (!res.ok) throw new Error('Failed to fetch papers');
   213→   213→      const data = await res.json();
   214→   214→      setFetchedItems(data.items || []);
   215→   215→    } catch (e) {
   216→   216→      setFeedError(e instanceof Error ? e.message : 'Failed to fetch');
   217→   217→      setFetchedItems([]);
   218→   218→    } finally {
   219→   219→      setFeedLoading(false);
   220→   220→    }
   221→   221→  }, []);
   222→   222→
   223→   223→  // Open feed panel
   224→   224→  const openFeed = useCallback((type: 'news' | 'papers') => {
   225→   225→    setShowFeedPanel(type);
   226→   226→    if (type === 'news') fetchNews(newsCategory);
   227→   227→    else fetchPapers(paperCategory);
   228→   228→  }, [newsCategory, paperCategory, fetchNews, fetchPapers]);
   229→   229→
   230→   230→  // Add fetched item to library
   231→   231→  const addFetchedItem = useCallback((item: FetchedItem) => {
   232→   232→    const newItem = onAddLibraryItem({
   233→   233→      title: item.title,
   234→   234→      author: item.author,
   235→   235→      type: item.type as 'pdf' | 'url' | 'rss',
   236→   236→      source: item.source,
   237→   237→      progressPercent: 0,
   238→   238→      status: 'unread',
   239→   239→      content: item.content ? `# ${item.title}\n\n${item.content}` : `# ${item.title}\n\n*Content from ${item.source}. Visit the original source for full text.*\n\nSource: ${item.url}`,
   240→   240→      chapterTitle: item.source,
   241→   241→    });
   242→   242→    return newItem;
   243→   243→  }, [onAddLibraryItem]);
   244→   244→
   245→   245→  // Handle file upload — sends file to API for real text extraction
   246→   246→  const [uploadingFile, setUploadingFile] = useState(false);
   247→   247→  const fileInputRef = useRef<HTMLInputElement>(null);
   248→   248→  const handleFileUpload = useCallback(async (e: React.ChangeEvent<HTMLInputElement>) => {
   249→   249→    const files = e.target.files;
   250→   250→    if (!files || files.length === 0) return;
   251→   251→    const file = files[0];
   252→   252→    const isEpub = file.name.endsWith('.epub');
   253→   253→    const isPdf = file.name.endsWith('.pdf');
   254→   254→    if (!isEpub && !isPdf) return;
   255→   255→
   256→   256→    setUploadingFile(true);
   257→   257→    try {
   258→   258→      // Send file to server for real text extraction
   259→   259→      const formData = new FormData();
   260→   260→      formData.append('file', file);
   261→   261→
   262→   262→      const res = await fetch('/api/reading/extract?XTransformPort=3000', {
   263→   263→        method: 'POST',
   264→   264→        body: formData,
   265→   265→      });
   266→   266→
   267→   267→      if (!res.ok) {
   268→   268→        throw new Error('Failed to extract content');
   269→   269→      }
   270→   270→
   271→   271→      const data = await res.json();
   272→   272→
   273→   273→      const newItem = onAddLibraryItem({
   274→   274→        title: data.title || file.name.replace(/\.(epub|pdf)$/, ''),
   275→   275→        author: data.author || 'Unknown',
   276→   276→        type: isEpub ? 'epub' : 'pdf',
   277→   277→        source: 'Uploaded file',
   278→   278→        progressPercent: 0,
   279→   279→        status: 'unread',
   280→   280→        content: data.content,
   281→   281→        totalPages: data.totalPages || 1,
   282→   282→        currentPage: 1,
   283→   283→        chapterTitle: isEpub ? 'Chapter 1' : 'Page 1',
   284→   284→      });
   285→   285→
   286→   286→      // Auto-select the newly added item
   287→   287→      setActiveItemId(newItem.id);
   288→   288→      setShowImport(false);
   289→   289→    } catch (err) {
   290→   290→      // Fallback: create item with error message
   291→   291→      const title = file.name.replace(/\.(epub|pdf)$/, '');
   292→   292→      onAddLibraryItem({
   293→   293→        title,
   294→   294→        author: 'Unknown',
   295→   295→        type: isEpub ? 'epub' : 'pdf',
   296→   296→        source: 'Uploaded file',
   297→   297→        progressPercent: 0,
   298→   298→        status: 'unread',
   299→   299→        content: `# ${title}\n\n*Failed to extract content from this file. The file may be corrupted or use a format that cannot be parsed.*\n\nFile: ${file.name}\nSize: ${(file.size / 1024 / 1024).toFixed(1)} MB`,
   300→   300→        totalPages: 1,
   301→   301→        currentPage: 1,
   302→   302→      });
   303→   303→      setShowImport(false);
   304→   304→    } finally {
   305→   305→      setUploadingFile(false);
   306→   306→      // Reset file input so the same file can be uploaded again
   307→   307→      if (fileInputRef.current) fileInputRef.current.value = '';
   308→   308→    }
   309→   309→  }, [onAddLibraryItem]);
   310→   310→
   311→   311→  // Auto-save progress on scroll
   312→   312→  const progressTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   313→   313→  const handleReaderScroll = useCallback(() => {
   314→   314→    if (!activeItemId || !readerRef.current) return;
   315→   315→    if (progressTimer.current) clearTimeout(progressTimer.current);
   316→   316→    progressTimer.current = setTimeout(() => {
   317→   317→      const el = readerRef.current;
   318→   318→      if (!el) return;
   319→   319→      const scrollPercent = Math.min(100, Math.round((el.scrollTop / (el.scrollHeight - el.clientHeight)) * 100));
   320→   320→      onUpdateLibraryItem(activeItemId, {
   321→   321→        progressPercent: scrollPercent,
   322→   322→        lastOpenedAt: new Date().toISOString(),
   323→   323→        status: scrollPercent >= 95 ? 'done' : 'reading',
   324→   324→      });
   325→   325→    }, 2000);
   326→   326→  }, [activeItemId, onUpdateLibraryItem]);
   327→   327→
   328→   328→  return (
   329→   329→    <div style={{ flex: 1, display: 'flex', flexDirection: 'column', overflow: 'hidden', background: 'var(--bg)' }}>
   330→   330→      {/* Topbar */}
   331→   331→      <div style={{
   332→   332→        height: 44, background: 'var(--bg1)', borderBottom: '1px solid var(--bd)',
   333→   333→        display: 'flex', alignItems: 'center', justifyContent: 'space-between', padding: '0 16px', flexShrink: 0,
   334→   334→      }}>
   335→   335→        <div style={{ display: 'flex', alignItems: 'center', gap: 8, fontSize: 12, color: 'var(--t3)' }}>
   336→   336→          <button onClick={onBack} style={{ background: 'transparent', border: 'none', color: 'var(--t3)', cursor: 'pointer', fontFamily: 'inherit', fontSize: 12, padding: 0 }}>dashboard</button>
   337→   337→          <span>/</span>
   338→   338→          <span style={{ color: 'var(--t1)', display: 'flex', alignItems: 'center', gap: 5 }}>
   339→   339→            <BookOpen size={13} color="var(--acc2)" /> reading
   340→   340→          </span>
   341→   341→        </div>
   342→   342→        <div style={{ display: 'flex', gap: 8 }}>
   343→   343→          <button onClick={() => openFeed('news')} style={{
   344→   344→            height: 28, padding: '0 12px',
   345→   345→            background: 'var(--bg2)', color: 'var(--t2)', border: '1px solid var(--bd2)', borderRadius: 4,
   346→   346→            fontSize: 11, fontFamily: 'inherit', cursor: 'pointer',
   347→   347→            display: 'flex', alignItems: 'center', gap: 5,
   348→   348→          }}>
   349→   349→            <Globe size={12} /> tech news
   350→   350→          </button>
   351→   351→          <button onClick={() => openFeed('papers')} style={{
   352→   352→            height: 28, padding: '0 12px',
   353→   353→            background: 'var(--bg2)', color: 'var(--t2)', border: '1px solid var(--bd2)', borderRadius: 4,
   354→   354→            fontSize: 11, fontFamily: 'inherit', cursor: 'pointer',
   355→   355→            display: 'flex', alignItems: 'center', gap: 5,
   356→   356→          }}>
   357→   357→            <FileText size={12} /> papers
   358→   358→          </button>
   359→   359→          <button onClick={() => setShowImport(true)} style={{
   360→   360→            height: 28, padding: '0 12px',
   361→   361→            background: 'var(--acc)', color: '#fff', border: 'none', borderRadius: 4,
   362→   362→            fontSize: 11, fontFamily: 'inherit', cursor: 'pointer', fontWeight: 600,
   363→   363→            display: 'flex', alignItems: 'center', gap: 5,
   364→   364→          }}>
   365→   365→            <Plus size={12} /> add source
   366→   366→          </button>
   367→   367→        </div>
   368→   368→      </div>
   369→   369→
   370→   370→      {/* 3-column layout */}
   371→   371→      <div style={{ flex: 1, display: 'flex', overflow: 'hidden' }}>
   372→   372→        {/* Library panel (220px) */}
   373→   373→        <div style={{
   374→   374→          width: 220, minWidth: 220, background: 'var(--bg1)', borderRight: '1px solid var(--bd)',
   375→   375→          display: 'flex', flexDirection: 'column', overflow: 'hidden',
   376→   376→        }}>
   377→   377→          <div style={{ padding: '10px 10px 6px', flexShrink: 0 }}>
   378→   378→            <div style={{
   379→   379→              display: 'flex', alignItems: 'center', gap: 8,
   380→   380→              background: 'var(--bg3)', border: '1px solid var(--bd2)', borderRadius: 4, padding: '0 10px', height: 30,
   381→   381→            }}>
   382→   382→              <Search size={13} color="var(--t3)" />
   383→   383→              <input value={search} onChange={(e) => setSearch(e.target.value)} placeholder="search library…"
   384→   384→                style={{ flex: 1, background: 'transparent', border: 'none', outline: 'none', color: 'var(--t1)', fontSize: 11, fontFamily: 'inherit', caretColor: 'var(--acc2)' }} />
   385→   385→            </div>
   386→   386→          </div>
   387→   387→          <div style={{ display: 'flex', padding: '0 10px 6px', gap: 2, flexShrink: 0 }}>
   388→   388→            {([
   389→   389→              { id: 'all', label: 'all' }, { id: 'books', label: 'books' },
   390→   390→              { id: 'papers', label: 'papers' }, { id: 'news', label: 'news' },
   391→   391→            ] as const).map((t) => (
   392→   392→              <button key={t.id} onClick={() => setTabFilter(t.id)} style={{
   393→   393→                flex: 1, padding: '3px 4px', background: 'transparent',
   394→   394→                border: 'none', borderBottom: tabFilter === t.id ? '2px solid var(--acc)' : '2px solid transparent',
   395→   395→                color: tabFilter === t.id ? 'var(--acc2)' : 'var(--t3)',
   396→   396→                fontSize: 9, fontFamily: 'inherit', cursor: 'pointer',
   397→   397→                textTransform: 'uppercase', letterSpacing: '0.06em', fontWeight: 600,
   398→   398→              }}>{t.label}</button>
   399→   399→            ))}
   400→   400→          </div>
   401→   401→          <div className="sb-scroll" style={{ flex: 1, overflowY: 'auto', padding: '4px 0' }}>
   402→   402→            {sections.reading.length > 0 && <LibrarySection label="currently reading" items={sections.reading} activeItemId={activeItemId} onSelect={setActiveItemId} />}
